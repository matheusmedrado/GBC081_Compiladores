int sinal(int x) {
    if (x > 0) {
        return 1;
    } else if (x < 0) {
        return -1;
    } else {
        return 0;
    }
}

    if (x > 0) {
        return 1;
    }
}

}

int depois(int x) {
    while (x > 0) {
        x = x - 1;
